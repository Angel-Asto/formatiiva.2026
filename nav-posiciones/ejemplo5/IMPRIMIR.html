3) Código CSS (css/styles.css)

/* ============================================================
   EJEMPLO 5 - MENU LATERAL VERTICAL (SIDEBAR)
   Posición del menú: COSTADO IZQUIERDO
   ============================================================ */

/* ---------- RESET: se pone SIEMPRE al principio ---------- */
* {                        /* el asterisco significa "TODOS los elementos" */
  margin: 0;               /* quita el espacio de AFUERA de la caja (distancia entre dos cajas de fruta) */
  padding: 0;              /* quita el relleno de ADENTRO (el cartón entre la caja y las manzanas) */
  box-sizing: border-box;  /* el ancho que pongo ya incluye el relleno: la caja no se "infla" */
}

body {                                        /* body = toda la página */
  font-family: Arial, Helvetica, sans-serif;  /* tipo de letra; si no existe Arial usa la siguiente */
}

a {                        /* todos los links */
  text-decoration: none;   /* les quita el subrayado */
}

ul {                       /* todas las listas */
  list-style: none;        /* les quita las viñetas (los puntitos) */
}

/* ---------- BOTONES ---------- */
.btn {                     /* el punto = "todo lo que tenga class='btn'" */
  display: inline-block;   /* se comporta como texto pero deja darle alto y relleno */
  padding: 12px 24px;      /* relleno: 12px arriba/abajo, 24px a izquierda/derecha */
  border-radius: 6px;      /* redondea las esquinas */
  font-weight: bold;       /* letra en negrita */
}

.btn--primary {            /* botón relleno */
  background-color: #f5a623;  /* fondo amarillo */
  color: #000;             /* letra negra */
}

.btn--outline {            /* botón solo con borde */
  border: 2px solid #fff;  /* borde de 2px, línea continua, blanco */
  color: #fff;             /* letra blanca */
}

.btn:hover {               /* :hover = cuando el mouse está encima */
  opacity: 0.85;           /* se ve un poquito transparente para avisar que es clickeable */
}

/* ---------- LA PAGINA EN 2 COLUMNAS ---------- */
body {
  display: flex;             /* las 2 cajas hijas del body (sidebar y contenido) quedan una AL LADO de la otra */
}
/* Para pasar el menú al lado DERECHO: agrega  flex-direction: row-reverse;  aquí arriba. */

/* ---------- LA BARRA LATERAL ---------- */
.sidebar {
  width: 240px;              /* ancho fijo de la barra lateral */
  min-height: 100vh;         /* alto mínimo = toda la pantalla (100vh = 100% del alto visible) */
  background-color: #0b3d5c; /* fondo azul oscuro */
  padding: 30px 20px;        /* relleno: 30px arriba/abajo y 20px a los lados */
}

.nav {
  display: flex;             /* activa flexbox dentro de la barra */
  flex-direction: column;    /* apila logo, links y botón UNO BAJO OTRO */
  align-items: center;       /* centra cada cosa en horizontal (recuerda: en column los ejes se invierten) */
  gap: 30px;                 /* 30px entre logo, links y botón */
}

.nav__logo img {           /* el logo */
  width: 150px;            /* aquí definimos el ANCHO porque la barra es angosta */
}

.nav__links {              /* la lista de links */
  display: flex;           /* activa flexbox */
  flex-direction: column;  /* links uno BAJO otro (vertical) en vez de en fila */
  gap: 20px;               /* 20px entre link y link */
  text-align: center;      /* texto centrado */
}

.nav__links a {            /* cada link */
  color: #fff;             /* letra blanca */
  font-weight: bold;       /* negrita */
}

.nav__links a:hover {      /* cuando el mouse pasa por encima */
  color: #f5a623;          /* cambia a amarillo */
}

/* ---------- EL CONTENIDO (a la derecha) ---------- */
.content {
  flex: 1;                   /* ocupa TODO el espacio que sobra al lado de la barra lateral */
}

.hero {
  background-color: #1d7a9e; /* fondo celeste */
  color: #fff;               /* texto blanco */
  text-align: center;        /* texto centrado */
  padding: 120px 20px;       /* relleno: 120px arriba/abajo y 20px a los lados */
  min-height: 100vh;         /* el bloque cubre toda la altura de la pantalla */
}

.hero h1 {                 /* título */
  font-size: 2.5rem;       /* tamaño grande */
  margin: 10px 0;          /* 10px arriba y abajo */
}

.hero h2 {                 /* subtítulo */
  margin-bottom: 15px;     /* separa del párrafo */
}

.hero__btns {              /* cajita de los 2 botones */
  display: flex;           /* botones en fila */
  justify-content: center; /* centrados en horizontal */
  gap: 15px;               /* 15px entre botones */
  margin-top: 25px;        /* separa del texto */
}
